CREATE TABLE addresses (
    addressID INTEGER PRIMARY KEY AUTOINCREMENT,
    addressName TEXT NOT NULL,
    latitude REAL NOT NULL,
    longitude REAL NOT NULL,
    city TEXT NOT NULL
);

CREATE TABLE person (
    personID TEXT PRIMARY KEY,
    firstName TEXT,
    lastName TEXT,
    birthYear INT,
    birthPlace TEXT,
    photo TEXT,
    notes TEXT
);

CREATE TABLE livedIn (
    personID TEXT,
    addressID INTEGER,
    familyID TEXT,
    relationToHead TEXT,
    year INT,
    occupation TEXT,
    PRIMARY KEY (personID, addressID),
    FOREIGN KEY (personID) REFERENCES person(personID),
    FOREIGN KEY (addressID) REFERENCES addresses(addressID)
);

SELECT DISTINCT a.addressID, a.addressName, a.latitude, a.longitude, a.city
FROM addresses a
JOIN livedIn l ON a.addressID = l.addressID
WHERE a.city = 'New York City' AND l.year = 1920;

SELECT l.addressID, COUNT(DISTINCT l.familyID) as familyCount
FROM livedIn l
JOIN addresses a ON l.addressID = a.addressID
WHERE a.city = 'New York City' AND l.year = 1920
GROUP BY a.addressID;

SELECT DISTINCT a.addressID, a.addressName, a.latitude, a.longitude, a.city, COUNT(DISTINCT l.familyID) AS familyCount
FROM addresses a
JOIN livedIn l ON a.addressID = l.addressID
WHERE a.city = 'New York City' AND l.year = 1920
GROUP BY a.addressID, a.addressName, a.latitude, a.longitude, a.city;
